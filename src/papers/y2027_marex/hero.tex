% \drone[label offset]{position}{color}{label}
\newcommand{\drone}[4][(0,0.42)]{%
  \begin{scope}[shift={#2}]
    \foreach \a in {45,135,225,315}
      \draw[#3!70!black, line width=0.7pt] (0,0) -- (\a:0.2);
    \foreach \a in {45,135,225,315}
      \filldraw[fill=#3!20, draw=#3!70!black, line width=0.4pt] (\a:0.2) circle (0.095);
    \filldraw[fill=#3, draw=#3!60!black, line width=0.4pt] (-0.08,-0.08) rectangle (0.08,0.08);
    \node[font=\scriptsize\bfseries, text=#3!80!black, fill=white, fill opacity=0.8, text opacity=1, inner sep=0.5pt, rounded corners=1pt] at #1 {#4};
  \end{scope}}
